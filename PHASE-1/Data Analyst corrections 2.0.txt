import java.util.*;

class Main {

    // ------------------- FUNCTIONS -------------------

    public static double mean(double[] data) {
        double sum = 0;
        for (int i = 0; i < data.length; i++) {
            sum += data[i];
        }
        return sum / data.length;
    }

    public static double median(double[] data) {
        double[] copy = Arrays.copyOf(data, data.length); // safe copy
        Arrays.sort(copy);

        int n = copy.length;
        if (n % 2 == 0) {
            return (copy[n/2 - 1] + copy[n/2]) / 2.0;
        } else {
            return copy[n/2];
        }
    }

    public static double max(double[] data) {
        double max = data[0];
        for (int i = 1; i < data.length; i++) {
            if (data[i] > max) {
                max = data[i];
            }
        }
        return max;
    }

    public static double min(double[] data) {
        double min = data[0];
        for (int i = 1; i < data.length; i++) {
            if (data[i] < min) {
                min = data[i];
            }
        }
        return min;
    }

    public static double variance(double[] data) {
        double mean = mean(data);
        double sum = 0;

        for (int i = 0; i < data.length; i++) {
            sum += Math.pow(data[i] - mean, 2);
        }
        return sum / data.length;
    }

    public static double stdDev(double[] data) {
        return Math.sqrt(variance(data));
    }

    public static void displayArray(double[] data) {
        System.out.print("Data: ");
        for (double num : data) {
            System.out.print(num + " ");
        }
        System.out.println();
    }

    // ------------------- MAIN METHOD -------------------

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of elements: ");
        int n = sc.nextInt();

        double[] data = new double[n];   // ARRAY CREATION

        System.out.println("Enter elements:");
        for (int i = 0; i < n; i++) {
            data[i] = sc.nextDouble();   // ARRAY INPUT
        }

        displayArray(data);

        System.out.println("\n--- Data Analysis ---");
        System.out.printf("Mean: %.2f\n", mean(data));
        System.out.printf("Median: %.2f\n", median(data));
        System.out.printf("Maximum: %.2f\n", max(data));
        System.out.printf("Minimum: %.2f\n", min(data));
        System.out.printf("Variance: %.2f\n", variance(data));
        System.out.printf("Standard Deviation: %.2f\n", stdDev(data));

        sc.close();
    }
}
